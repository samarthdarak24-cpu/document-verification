# DocuTrace backend dependencies.
#
# The DocuTrace application is a pure-algorithms verification service: every
# text/duplicate algorithm lives in ``backend/app/algorithms`` (implemented from
# scratch for the DAA project) and only *wraps* well-tested third-party
# primitives for I/O, persistence and rendering (PyMuPDF, python-docx, PIL,
# OpenCV, ReportLab, SQLAlchemy).
#
# Tesseract OCR is invoked through ``pytesseract`` but the Tesseract *binary*
# itself is a system dependency (apt/brew/choco install tesseract-ocr) that is
# intentionally NOT declared here. The extraction layer detects its absence
# (``pytesseract.get_tesseract_version()`` raising) and degrades gracefully with
# a recorded warning instead of crashing.

# --- Web framework / server -------------------------------------------------
fastapi==0.142.2
uvicorn[standard]==0.54.0
starlette>=0.40.0

# --- Persistence ------------------------------------------------------------
SQLAlchemy>=2.0,<3.0
alembic==1.20.0

# --- Validation / serialisation --------------------------------------------
pydantic>=2.6,<3.0
email-validator==2.3.0
python-dotenv==1.0.1      # loads backend/.env so TESSERACT_CMD is present (used in app/main.py)

# --- Document parsing & rendering ------------------------------------------
pymupdf==1.28.2            # PDF text extraction + page rendering
python-docx==1.2.0         # DOCX paragraphs / headings / tables
pillow==12.3.0             # image open, EXIF transpose, grayscale, autocontrast
opencv-python-headless==5.0.0.93  # optional image preprocessing (headless: no GUI)
pytesseract==0.3.13        # OCR client (needs the Tesseract binary at runtime)
reportlab==5.0.1           # verification / comparison PDF reports
python-multipart==0.0.32   # multipart/form-data uploads
fast-diff-match-patch>=2.1.0  # fast diff engine for pdf-diff
lxml>=5.0.0                # XML parser for poppler/pdf-diff

# --- Testing ----------------------------------------------------------------
pytest==9.1.1
